add_library(ImageLib
  ${PROJECT_SOURCE_DIR}/src/image.cpp
)

target_include_directories(ImageLib
  PUBLIC
    ${PROJECT_SOURCE_DIR}/src
    ${ZLIB_INCLUDE_DIRS}
)

target_link_libraries(ImageLib
  PUBLIC
    ByteReader
    ZLIB::ZLIB
    SDL3::SDL3
)

  bytereader_tests.cpp
  paeth_tests.cpp
  image_tests.cpp
)

  PRIVATE
    ${PROJECT_SOURCE_DIR}/src
)

  PRIVATE
    ImageLib
    GTest::gtest_main
    ByteReader
)
